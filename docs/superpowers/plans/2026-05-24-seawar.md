# SeaWar 海战棋 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a LAN multiplayer Battleship (SeaWar) game using TypeScript + Bun, with room-based matchmaking, manual/random ship placement, and a power-up shell system.

**Architecture:** Bun single-process full-stack. `server.ts` serves static files via `Bun.serve()` and handles WebSocket game logic in-memory. Frontend is a single-page DOM app using CSS Grid for the board.

**Tech Stack:** Bun, TypeScript, WebSocket, DOM CSS Grid. Zero external dependencies.

import { Ship, Coord, BOARD_SIZE } from '../types';

function coordKey(c: Coord): string {
  return `${c.x},${c.y}`;
}

function isAdjacent(a: Coord, b: Coord): boolean {
  return Math.abs(a.x - b.x) <= 1 && Math.abs(a.y - b.y) <= 1 && !(a.x === b.x && a.y === b.y);
}

export function validateShipPlacement(ships: Ship[]): { valid: boolean; error?: string } {
  const occupied = new Set<string>();
  const allCoords: Coord[] = [];

  for (const ship of ships) {
    // Check size matches coords
    if (ship.coords.length !== ship.size) {
      return { valid: false, error: `Ship ${ship.id} size mismatch` };
    }

    // Check bounds
    for (const c of ship.coords) {
      if (c.x < 0 || c.x >= BOARD_SIZE || c.y < 0 || c.y >= BOARD_SIZE) {
        return { valid: false, error: `Ship ${ship.id} out of bounds` };
      }
    }

    // Check contiguous (horizontal or vertical)
    const xs = ship.coords.map(c => c.x);
    const ys = ship.coords.map(c => c.y);
    const isHorizontal = ys.every(y => y === ys[0]);
    const isVertical = xs.every(x => x === xs[0]);
    if (!isHorizontal && !isVertical) {
      return { valid: false, error: `Ship ${ship.id} not straight` };
    }
    const sorted = isHorizontal
      ? [...ship.coords].sort((a, b) => a.x - b.x)
      : [...ship.coords].sort((a, b) => a.y - b.y);
    for (let i = 1; i < sorted.length; i++) {
      const dist = isHorizontal ? sorted[i].x - sorted[i - 1].x : sorted[i].y - sorted[i - 1].y;
      if (dist !== 1) {
        return { valid: false, error: `Ship ${ship.id} not contiguous` };
      }
    }

    // Check overlap
    for (const c of ship.coords) {
      const key = coordKey(c);
      if (occupied.has(key)) {
        return { valid: false, error: `Ship ${ship.id} overlaps` };
      }
      occupied.add(key);
      allCoords.push(c);
    }
  }

  // Check spacing (at least 1 cell apart, including diagonals)
  for (let i = 0; i < allCoords.length; i++) {
    for (let j = i + 1; j < allCoords.length; j++) {
      if (isAdjacent(allCoords[i], allCoords[j])) {
        return { valid: false, error: 'Ships too close' };
      }
    }
  }

  return { valid: true };
}

export function generateRandomShips(): Ship[] {
  const sizes = [5, 4, 3, 3, 2];
  const ships: Ship[] = [];
  const occupied = new Set<string>();

  function getNeighbors(c: Coord): Coord[] {
    const neighbors: Coord[] = [];
    for (let dx = -1; dx <= 1; dx++) {
      for (let dy = -1; dy <= 1; dy++) {
        if (dx === 0 && dy === 0) continue;
        const nx = c.x + dx, ny = c.y + dy;
        if (nx >= 0 && nx < BOARD_SIZE && ny >= 0 && ny < BOARD_SIZE) {
          neighbors.push({ x: nx, y: ny });
        }
      }
    }
    return neighbors;
  }

  for (let i = 0; i < sizes.length; i++) {
    const size = sizes[i];
    let placed = false;
    let attempts = 0;

    while (!placed && attempts < 1000) {
      attempts++;
      const horizontal = Math.random() < 0.5;
      const x = Math.floor(Math.random() * (horizontal ? BOARD_SIZE - size + 1 : BOARD_SIZE));
      const y = Math.floor(Math.random() * (horizontal ? BOARD_SIZE : BOARD_SIZE - size + 1));

      const coords: Coord[] = [];
      for (let j = 0; j < size; j++) {
        coords.push(horizontal ? { x: x + j, y } : { x, y: y + j });
      }

      // Check bounds and overlap
      const keys = coords.map(coordKey);
      if (keys.some(k => occupied.has(k))) continue;

      // Check spacing
      const tooClose = coords.some(c =>
        getNeighbors(c).some(n => occupied.has(coordKey(n)))
      );
      if (tooClose) continue;

      // Place ship
      for (const c of coords) {
        occupied.add(coordKey(c));
        for (const n of getNeighbors(c)) {
          occupied.add(coordKey(n));
        }
      }

      ships.push({
        id: `ship-${i}`,
        size,
        coords,
        hits: [],
        sunk: false,
      });
      placed = true;
    }

    if (!placed) {
      // Restart generation if failed
      return generateRandomShips();
    }
  }

  return ships;
}

import { Coord, BOARD_SIZE } from '../types';

function inBounds(c: Coord): boolean {
  return c.x >= 0 && c.x < BOARD_SIZE && c.y >= 0 && c.y < BOARD_SIZE;
}

export function resolveCross(center: Coord): Coord[] {
  const coords = [center];
  const dirs = [{ x: 0, y: -1 }, { x: 0, y: 1 }, { x: -1, y: 0 }, { x: 1, y: 0 }];
  for (const d of dirs) {
    const c = { x: center.x + d.x, y: center.y + d.y };
    if (inBounds(c)) coords.push(c);
  }
  return coords;
}

export function resolveMulti(center: Coord, allShots: Set<string>): Coord[] {
  const coords = [center];
  const available: Coord[] = [];
  for (let x = 0; x < BOARD_SIZE; x++) {
    for (let y = 0; y < BOARD_SIZE; y++) {
      const key = `${x},${y}`;
      if (!allShots.has(key)) {
        available.push({ x, y });
      }
    }
  }
  // Exclude center
  const filtered = available.filter(c => c.x !== center.x || c.y !== center.y);
  // Shuffle and pick 4
  for (let i = filtered.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [filtered[i], filtered[j]] = [filtered[j], filtered[i]];
  }
  coords.push(...filtered.slice(0, 4));
  return coords;
}

export function resolveNuke(center: Coord): Coord[] {
  const coords: Coord[] = [];
  // Diamond shape: rows have 1,3,5,3,1 cells
  const pattern = [
    { dx: 0, dy: -2, width: 1 },
    { dx: -1, dy: -1, width: 3 },
    { dx: -2, dy: 0, width: 5 },
    { dx: -1, dy: 1, width: 3 },
    { dx: 0, dy: 2, width: 1 },
  ];
  for (const row of pattern) {
    for (let i = 0; i < row.width; i++) {
      const c = { x: center.x + row.dx + i, y: center.y + row.dy };
      if (inBounds(c)) coords.push(c);
    }
  }
  return coords;
}

import { Board, ITEM_SPAWN_CHANCE, ITEM_SPAWN_MIN_TURN, BOARD_SIZE } from '../types';

export function shouldSpawnItem(turnCount: number): boolean {
  if (turnCount < ITEM_SPAWN_MIN_TURN) return false;
  return Math.random() < ITEM_SPAWN_CHANCE;
}

export function spawnItem(board: Board): { x: number; y: number } | null {
  const available: { x: number; y: number }[] = [];
  for (let x = 0; x < BOARD_SIZE; x++) {
    for (let y = 0; y < BOARD_SIZE; y++) {
      const key = `${x},${y}`;
      if (!board.shots.has(key)) {
        available.push({ x, y });
      }
    }
  }
  if (available.length === 0) return null;
  const idx = Math.floor(Math.random() * available.length);
  const pos = available[idx];
  board.items.add(`${pos.x},${pos.y}`);
  return pos;
}

import { Board, Ship, Coord, ShellType, SHELL_TYPES, STARTING_SHELLS } from '../types';
import { resolveCross, resolveMulti, resolveNuke } from './shell-resolver';
import { spawnItem, shouldSpawnItem } from './item-spawner';

function key(c: Coord): string {
  return `${c.x},${c.y}`;
}

export function createBoard(): Board {
  return { ships: [], shots: new Map(), items: new Set() };
}

export function createPlayer(id: string, name: string) {
  return {
    id,
    name,
    board: createBoard(),
    ready: false,
    inventory: [...STARTING_SHELLS],
  };
}

export function placeShips(board: Board, ships: Ship[]): void {
  board.ships = ships;
}

export function checkHit(board: Board, coord: Coord): { hit: boolean; ship?: Ship } {
  const k = key(coord);
  for (const ship of board.ships) {
    if (ship.coords.some(c => key(c) === k)) {
      return { hit: true, ship };
    }
  }
  return { hit: false };
}

export function fire(board: Board, coord: Coord): {
  result: 'hit' | 'miss';
  shipSunk?: Ship;
  gotShell?: boolean;
  shellType?: ShellType;
} {
  const k = key(coord);
  if (board.shots.has(k)) {
    throw new Error('Already fired at this location');
  }

  const { hit, ship } = checkHit(board, coord);
  board.shots.set(k, hit ? 'hit' : 'miss');

  let shipSunk: Ship | undefined;
  let gotShell = false;
  let shellType: ShellType | undefined;

  if (hit && ship) {
    ship.hits.push(coord);
    if (ship.hits.length === ship.size) {
      ship.sunk = true;
      shipSunk = ship;
    }
  }

  // Check if hit an item
  if (board.items.has(k)) {
    board.items.delete(k);
    gotShell = true;
    shellType = SHELL_TYPES[Math.floor(Math.random() * SHELL_TYPES.length)];
  }

  return { result: hit ? 'hit' : 'miss', shipSunk, gotShell, shellType };
}

export function useShell(
  board: Board,
  shellType: ShellType,
  coord: Coord
): {
  targets: { coord: Coord; result: 'hit' | 'miss'; shipSunk?: Ship }[];
  gotShell?: boolean;
  newShellType?: ShellType;
} {
  let coords: Coord[];
  const allShots = new Set(board.shots.keys());

  switch (shellType) {
    case 'cross':
      coords = resolveCross(coord);
      break;
    case 'multi':
      coords = resolveMulti(coord, allShots);
      break;
    case 'nuke':
      coords = resolveNuke(coord);
      break;
    default:
      throw new Error('Unknown shell type');
  }

  const targets: { coord: Coord; result: 'hit' | 'miss'; shipSunk?: Ship }[] = [];
  let gotShell = false;
  let newShellType: ShellType | undefined;

  for (const c of coords) {
    const k = key(c);
    if (board.shots.has(k)) continue;

    const { hit, ship } = checkHit(board, c);
    board.shots.set(k, hit ? 'hit' : 'miss');

    let shipSunk: Ship | undefined;
    if (hit && ship) {
      ship.hits.push(c);
      if (ship.hits.length === ship.size) {
        ship.sunk = true;
        shipSunk = ship;
      }
    }

    targets.push({ coord: c, result: hit ? 'hit' : 'miss', shipSunk });

    // Check item
    if (board.items.has(k)) {
      board.items.delete(k);
      gotShell = true;
      newShellType = SHELL_TYPES[Math.floor(Math.random() * SHELL_TYPES.length)];
    }
  }

  return { targets, gotShell, newShellType };
}

export function checkWin(board: Board): boolean {
  return board.ships.length > 0 && board.ships.every(s => s.sunk);
}

export function getRandomShellType(): ShellType {
  return SHELL_TYPES[Math.floor(Math.random() * SHELL_TYPES.length)];
}

export { shouldSpawnItem, spawnItem };
```

import { Room, Player, RoomPhase, STARTING_SHELLS } from '../types';
import { createPlayer, createBoard, placeShips } from './game-logic';

const rooms = new Map<string, Room>();

function generateRoomId(): string {
  return Math.floor(100000 + Math.random() * 900000).toString();
}

export function createRoom(playerName: string, ws: any): { room: Room; player: Player } {
  let roomId = generateRoomId();
  while (rooms.has(roomId)) {
    roomId = generateRoomId();
  }

  const player = createPlayer(crypto.randomUUID(), playerName);
  (player as any).ws = ws;

  const room: Room = {
    id: roomId,
    players: [player, null],
    phase: 'lobby',
    currentTurn: '',
    winner: null,
    turnCount: 0,
    playAgainVotes: new Set(),
    createdAt: Date.now(),
  };

  rooms.set(roomId, room);
  return { room, player };
}

export function joinRoom(roomId: string, playerName: string, ws: any): { room: Room; player: Player } | null {
  const room = rooms.get(roomId);
  if (!room) return null;
  if (room.players[1]) return null;

  const player = createPlayer(crypto.randomUUID(), playerName);
  (player as any).ws = ws;
  room.players[1] = player;

  return { room, player };
}

export function getRoom(roomId: string): Room | undefined {
  return rooms.get(roomId);
}

export function removeRoom(roomId: string): void {
  rooms.delete(roomId);
}

export function leaveRoom(room: Room, playerId: string): void {
  if (room.players[0]?.id === playerId) {
    room.players[0] = null;
  } else if (room.players[1]?.id === playerId) {
    room.players[1] = null;
  }
  if (!room.players[0] && !room.players[1]) {
    removeRoom(room.id);
  }
}

export function broadcast(room: Room, message: object, excludePlayerId?: string): void {
  for (const p of room.players) {
    if (p && p.id !== excludePlayerId) {
      const ws = (p as any).ws;
      if (ws && ws.readyState === 1) {
        ws.send(JSON.stringify(message));
      }
    }
  }
}

export function getOpponent(room: Room, playerId: string): Player | null {
  if (room.players[0]?.id === playerId) return room.players[1];
  if (room.players[1]?.id === playerId) return room.players[0];
  return null;
}

export function resetRoomForRestart(room: Room): void {
  room.phase = 'placement';
  room.currentTurn = '';
  room.winner = null;
  room.turnCount = 0;
  room.playAgainVotes.clear();

  for (const p of room.players) {
    if (p) {
      p.board = createBoard();
      p.ready = false;
      p.inventory = [...STARTING_SHELLS];
    }
  }
}

// Cleanup old rooms every 5 minutes
setInterval(() => {
  const now = Date.now();
  for (const [id, room] of rooms) {
    if (now - room.createdAt > 30 * 60 * 1000) {
      rooms.delete(id);
    }
  }
}, 5 * 60 * 1000);
```

  const ctx = wsToRoom.get(ws);
  if (!ctx) return;
  const { room, player } = ctx;
  leaveRoom(room, player.id);
  wsToRoom.delete(ws);

  if (room.players[0] || room.players[1]) {
    broadcast(room, { type: 'OPPONENT_LEFT' });
  }
}

function handleReady(ws: ServerWebSocket): void {
  const ctx = wsToRoom.get(ws);
  if (!ctx) return;
  ctx.player.ready = true;
  sendRoomState(ctx.room);
}

function handlePlaceShips(ws: ServerWebSocket, msg: Extract<ClientMessage, { type: 'PLACE_SHIPS' }>): void {
  const ctx = wsToRoom.get(ws);
  if (!ctx || ctx.room.phase !== 'placement') return;

  const validation = validateShipPlacement(msg.ships);
  if (!validation.valid) {
    send(ws, { type: 'ERROR', message: validation.error || 'Invalid placement' });
    return;
  }

  placeShips(ctx.player.board, msg.ships);
  ctx.player.ready = true;

  checkGameStart(ctx.room);
}

function handlePlaceShipsAuto(ws: ServerWebSocket): void {
  const ctx = wsToRoom.get(ws);
  if (!ctx || ctx.room.phase !== 'placement') return;

  const ships = generateRandomShips();
  placeShips(ctx.player.board, ships);
  ctx.player.ready = true;

  checkGameStart(ctx.room);
}

function checkGameStart(room: Room): void {
  const bothReady = room.players[0]?.ready && room.players[1]?.ready;
  if (!bothReady) return;

  room.phase = 'battle';
  room.currentTurn = room.players[0]!.id;
  room.turnCount = 0;

  // Send initial inventories
  for (const p of room.players) {
    if (p) {
      const ws = (p as any).ws as ServerWebSocket;
      send(ws, { type: 'INVENTORY_UPDATE', shells: p.inventory });
    }
  }

  broadcast(room, { type: 'GAME_START', firstTurn: room.currentTurn });
}

function handleFire(ws: ServerWebSocket, msg: Extract<ClientMessage, { type: 'FIRE' }>): void {
  const ctx = wsToRoom.get(ws);
  if (!ctx || ctx.room.phase !== 'battle') return;
  const { room, player } = ctx;

  if (room.currentTurn !== player.id) {
    send(ws, { type: 'ERROR', message: 'Not your turn' });
    return;
  }

  const opponent = getOpponent(room, player.id);
  if (!opponent) return;

  try {
    const result = fire(opponent.board, { x: msg.x, y: msg.y });

    // Send result to both players
    const fireResult: ServerMessage = {
      type: 'FIRE_RESULT',
      x: msg.x,
      y: msg.y,
      result: result.result,
      shipSunk: result.shipSunk,
      gotShell: result.gotShell,
      shellType: result.shellType,
    };
    broadcast(room, fireResult);

    // Update inventory if got shell
    if (result.gotShell && result.shellType) {
      player.inventory.push(result.shellType);
      send(ws, { type: 'INVENTORY_UPDATE', shells: player.inventory });
    }

    // Check win
    if (checkWin(opponent.board)) {
      room.phase = 'ended';
      room.winner = player.id;
      broadcast(room, { type: 'GAME_OVER', winner: player.id, reason: 'all_ships_sunk' });
      return;
    }

    // Handle turn change
    if (result.result === 'miss') {
      room.turnCount++;
      room.currentTurn = opponent.id;
      broadcast(room, { type: 'TURN_CHANGE', currentTurn: room.currentTurn });

      // Try spawn items after turn change
      trySpawnItems(room);
    }
  } catch (e: any) {
    send(ws, { type: 'ERROR', message: e.message });
  }
}

function handleUseShell(ws: ServerWebSocket, msg: Extract<ClientMessage, { type: 'USE_SHELL' }>): void {
  const ctx = wsToRoom.get(ws);
  if (!ctx || ctx.room.phase !== 'battle') return;
  const { room, player } = ctx;

  if (room.currentTurn !== player.id) {
    send(ws, { type: 'ERROR', message: 'Not your turn' });
    return;
  }

  const shellIndex = player.inventory.indexOf(msg.shellType);
  if (shellIndex === -1) {
    send(ws, { type: 'ERROR', message: 'Shell not in inventory' });
    return;
  }

  const opponent = getOpponent(room, player.id);
  if (!opponent) return;

  // Remove shell from inventory
  player.inventory.splice(shellIndex, 1);
  send(ws, { type: 'INVENTORY_UPDATE', shells: player.inventory });

  const result = useShell(opponent.board, msg.shellType as ShellType, { x: msg.x, y: msg.y });

  // Send shell result
  const shellResult: ServerMessage = {
    type: 'SHELL_RESULT',
    shellType: msg.shellType,
    targets: result.targets.map(t => ({
      x: t.coord.x,
      y: t.coord.y,
      result: t.result,
      shipSunk: t.shipSunk,
    })),
    gotShell: result.gotShell,
    newShellType: result.newShellType,
  };
  broadcast(room, shellResult);

  // Update inventory if got shell from item
  if (result.gotShell && result.newShellType) {
    player.inventory.push(result.newShellType);
    send(ws, { type: 'INVENTORY_UPDATE', shells: player.inventory });
  }

  // Check win
  if (checkWin(opponent.board)) {
    room.phase = 'ended';
    room.winner = player.id;
    broadcast(room, { type: 'GAME_OVER', winner: player.id, reason: 'all_ships_sunk' });
    return;
  }

  // Count as a turn - shells end the turn (you don't continue after using a shell)
  room.turnCount++;
  room.currentTurn = opponent.id;
  broadcast(room, { type: 'TURN_CHANGE', currentTurn: room.currentTurn });

  // Try spawn items
  trySpawnItems(room);
}

function trySpawnItems(room: Room): void {
  if (!shouldSpawnItem(room.turnCount)) return;

  const positions: { playerId: string; x: number; y: number }[] = [];

  for (const p of room.players) {
    if (p) {
      const pos = spawnItem(p.board);
      if (pos) {
        positions.push({ playerId: p.id, x: pos.x, y: pos.y });
      }
    }
  }

  if (positions.length > 0) {
    broadcast(room, { type: 'ITEM_SPAWNED', positions });
  }
}

function handlePlayAgain(ws: ServerWebSocket): void {
  const ctx = wsToRoom.get(ws);
  if (!ctx || ctx.room.phase !== 'ended') return;
  const { room, player } = ctx;

  room.playAgainVotes.add(player.id);

  if (room.playAgainVotes.size === 2) {
    resetRoomForRestart(room);
    broadcast(room, { type: 'RESTART_READY' });
    sendRoomState(room);
  }
}

function sendRoomState(room: Room): void {
  const msg: ServerMessage = {
    type: 'ROOM_STATE',
    roomId: room.id,
    players: getPlayerInfos(room),
    phase: room.phase,
  };
  broadcast(room, msg);
}

function getPlayerInfos(room: Room) {
  return room.players
    .filter((p): p is Player => p !== null)
    .map(p => ({ id: p.id, name: p.name, ready: p.ready }));
}

function isValidPlacement(coords: Coord[]): boolean {
  // Check bounds
  if (coords.some(c => c.x < 0 || c.x >= 10 || c.y < 0 || c.y >= 10)) return false;
  // Check overlap and spacing
  const allPlaced = placedShips.flatMap(s => s.coords);
  for (const c of coords) {
    for (const p of allPlaced) {

      if (Math.abs(p.x - c.x) <= 1 && Math.abs(p.y - c.y) <= 1) return false;
    }
  }
  return true;
}

function fireAt(x: number, y: number) {
  if (!isMyTurn) return;
  const cell = getEnemyCell(x, y);

    selectedShell = null;

    send({ type: 'FIRE', x, y });

function handleShellResult(msg: Extract<ServerMessage, { type: 'SHELL_RESULT' }>) {
  for (const t of msg.targets) {

    if (t.shipSunk) {
      // Mark all cells of sunk ship
      for (const c of t.shipSunk.coords) {
        const sunkCell = getEnemyCell(c.x, c.y);
        sunkCell.classList.add('sunk');
      }
    }
  }
}

      }
    }
  }
}
